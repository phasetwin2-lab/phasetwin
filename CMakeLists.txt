cmake_minimum_required(VERSION 3.22)
project(PhaseTwin VERSION 1.9.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PHASETWIN_DSP_ONLY "Build DSP tests without JUCE" OFF)
include(CTest)
add_executable(phasetwin_tests tests/dsp_tests.cpp)
target_include_directories(phasetwin_tests PRIVATE Source)
add_test(NAME dsp COMMAND phasetwin_tests)
find_package(Threads REQUIRED)
add_executable(phasetwin_regressions tests/alignment_regressions.cpp)
target_include_directories(phasetwin_regressions PRIVATE Source)
target_link_libraries(phasetwin_regressions PRIVATE Threads::Threads)
add_test(NAME alignment_regressions COMMAND phasetwin_regressions)
add_executable(phasetwin_workflow_tests tests/workflow_tests.cpp)
target_include_directories(phasetwin_workflow_tests PRIVATE Source)
add_test(NAME workflow COMMAND phasetwin_workflow_tests)
add_executable(phasetwin_learning_tests tests/learning_tests.cpp)
target_include_directories(phasetwin_learning_tests PRIVATE Source)
add_test(NAME learning COMMAND phasetwin_learning_tests)
add_executable(phasetwin_kick_tests tests/kick_tests.cpp)
target_include_directories(phasetwin_kick_tests PRIVATE Source)
add_test(NAME kick_learning COMMAND phasetwin_kick_tests)
add_executable(phasetwin_scope_trigger_tests tests/scope_trigger_tests.cpp)
target_include_directories(phasetwin_scope_trigger_tests PRIVATE Source)
add_test(NAME scope_trigger COMMAND phasetwin_scope_trigger_tests)
add_executable(phasetwin_delay_quality_tests tests/delay_quality_tests.cpp)
target_include_directories(phasetwin_delay_quality_tests PRIVATE Source)
add_test(NAME delay_quality COMMAND phasetwin_delay_quality_tests)
add_executable(phasetwin_microphone_tests tests/microphone_tests.cpp)
target_include_directories(phasetwin_microphone_tests PRIVATE Source)
add_test(NAME microphone_alignment COMMAND phasetwin_microphone_tests)
add_executable(phasetwin_view_tests tests/view_tests.cpp)
target_include_directories(phasetwin_view_tests PRIVATE Source)
add_test(NAME view_preferences COMMAND phasetwin_view_tests)
add_executable(phasetwin_ducking_tests tests/ducking_tests.cpp)
target_include_directories(phasetwin_ducking_tests PRIVATE Source)
add_test(NAME sidechain_ducking COMMAND phasetwin_ducking_tests)
if(PHASETWIN_DSP_ONLY)
  return()
endif()
set(JUCE_DIR "" CACHE PATH "Path to JUCE source checkout")
if(NOT EXISTS "${JUCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Download JUCE 8 and configure with -DJUCE_DIR=/path/to/JUCE")
endif()
add_subdirectory("${JUCE_DIR}" juce)
juce_add_plugin(PhaseTwin COMPANY_NAME "PhaseTwin" PLUGIN_MANUFACTURER_CODE PhTw
    PLUGIN_CODE Pt01 FORMATS VST3 PRODUCT_NAME "PhaseTwin"
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Tools)
juce_generate_juce_header(PhaseTwin)
target_sources(PhaseTwin PRIVATE Source/PluginProcessor.cpp Source/PluginProcessor.h Source/Alignment.h Source/AudioEngine.h Source/ScopeData.h Source/PluginEditor.h Source/ScopeHistory.h Source/Learning.h Source/Ducking.h)
target_compile_definitions(PhaseTwin PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(PhaseTwin PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
juce_add_console_app(phasetwin_plugin_tests PRODUCT_NAME "PhaseTwin integration tests")
juce_generate_juce_header(phasetwin_plugin_tests)
target_sources(phasetwin_plugin_tests PRIVATE tests/plugin_tests.cpp Source/PluginProcessor.cpp)
target_include_directories(phasetwin_plugin_tests PRIVATE Source)
target_compile_definitions(phasetwin_plugin_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(phasetwin_plugin_tests PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
add_test(NAME native_plugin COMMAND phasetwin_plugin_tests)
